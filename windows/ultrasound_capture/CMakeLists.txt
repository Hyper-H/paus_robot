cmake_minimum_required(VERSION 3.20)
project(paus_ultrasound_capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(capture_core core.cpp editor_core.cpp)
target_include_directories(capture_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
enable_testing()
add_executable(capture_core_test tests/core_test.cpp)
target_link_libraries(capture_core_test PRIVATE capture_core)
add_test(NAME capture_core_test COMMAND capture_core_test)
add_executable(editor_core_test tests/editor_core_test.cpp)
target_link_libraries(editor_core_test PRIVATE capture_core)
add_test(NAME editor_core_test COMMAND editor_core_test)

if(WIN32)
  file(GLOB SDK_PROJECTIONS "C:/Program Files (x86)/Windows Kits/10/Include/*/cppwinrt")
  list(SORT SDK_PROJECTIONS COMPARE NATURAL ORDER DESCENDING)
  find_path(CPPWINRT_INCLUDE_DIR winrt/base.h
    HINTS "$ENV{CPPWINRT_INCLUDE_DIR}" ${SDK_PROJECTIONS})
  if(NOT CPPWINRT_INCLUDE_DIR)
    message(FATAL_ERROR "Install Windows SDK C++/WinRT or set CPPWINRT_INCLUDE_DIR")
  endif()
  add_library(capture_windows capture_win.cpp profile_win.cpp)
  target_link_libraries(capture_windows PUBLIC capture_core windowsapp runtimeobject d3d11 dxgi ws2_32 shell32 ole32 user32)
  target_include_directories(capture_windows PUBLIC ${CPPWINRT_INCLUDE_DIR})
  add_executable(PausUltrasound WIN32 desktop.cpp icons_win.cpp)
  target_link_libraries(PausUltrasound PRIVATE capture_windows gdi32 comdlg32 comctl32 msimg32)
  target_link_libraries(capture_core_test PRIVATE capture_windows)
  foreach(target capture_core capture_core_test editor_core_test capture_windows PausUltrasound)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /await)
    endif()
  endforeach()
  install(TARGETS PausUltrasound RUNTIME DESTINATION .)
  install(FILES assets/icons/LICENSE DESTINATION licenses RENAME lucide-license.txt)
else()
  target_compile_options(capture_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(capture_core_test PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(editor_core_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
